% Figure 1 --- The Three-Machine Test.
% Editable TikZ source. Sized for a two-column figure* (usable width
% ~17.5cm). Shows the three environments, what is transmitted, what is
% WITHHELD from M3, and which evidence each machine contributes.
% Identity verification (hash recomputation) is drawn distinctly from
% acceptance execution (obligations re-run).
\begin{tikzpicture}[
  font=\footnotesize,
  x=1cm, y=1cm,
  mach/.style={draw, thick, rounded corners=2pt, align=left,
               inner sep=4pt, fill=black!3},
  pass/.style={draw, rounded corners=1.5pt, align=center,
               inner sep=3pt, fill=black!7},
  flow/.style={-{Latex[length=1.8mm]}, thick},
  cut/.style={-{Latex[length=1.8mm]}, thick, dashed, draw=black!55},
  ev/.style={font=\scriptsize, align=left},
]

% ------------------------------------------------------------ M1
\node[mach, text width=4.3cm, anchor=west] (m1) at (0,0) {%
  \textbf{M1\quad Production}\\[2pt]
  \scriptsize
  specification $S$: acceptance programs,\\
  fixtures, pinned inputs, recipe, limits\\[2pt]
  implementation $I$ \ \emph{(excluded from $\rho$)}\\[3pt]
  \rule{4.0cm}{0.3pt}\\[2pt]
  seal \ $\rho \leftarrow H(\operatorname{canon}(\Com(S)))$\\
  audit \ $A_S(I)$ earned};

% ------------------------------------------------- transmitted object
\node[pass, text width=2.7cm, anchor=west] (claim) at (5.4,0) {%
  \scriptsize\textbf{sealed claim}\\[1pt]
  $\Com(S)$\\ manifest $\rho$\\ $I$};

\draw[flow] (m1.east) -- (claim.west);

% ------------------------------------------------------------ M2
\node[mach, text width=6.4cm, anchor=west] (m2) at (9.6,1.55) {%
  \textbf{M2\quad Transfer} \hfill \scriptsize receives \emph{all} bytes\\[2pt]
  \scriptsize
  $\rho$ recomputed from the received bytes alone;\\
  every obligation re-executed. Establishes that the\\
  record is self-carrying, independent of the host.};

% ------------------------------------------------------------ M3
\node[mach, text width=6.4cm, anchor=west] (m3) at (9.6,-1.55) {%
  \textbf{M3\quad Reconstruction} \hfill \scriptsize receives $\Com(S)$ \emph{only}\\[2pt]
  \scriptsize
  producer $P$ emits a new $I'$; $\rho'$ sealed from $I'$;\\
  every obligation re-executed. Establishes that the\\
  disclosed requirements \emph{determine} the software.};

\draw[flow] (claim.east) -- ++(0.5,0) |- (m2.west);
\draw[flow] (claim.east) -- ++(0.5,0) |- (m3.west);

% ------------------------------------------- the withheld implementation
\node[ev, anchor=north] (wh) at (6.75,-1.45)
  {\itshape $I$ withheld\\\itshape from M3};
\draw[cut] (claim.south) -- (wh.north);
\node at (6.75,-1.12) {\scriptsize$\times$};

% ------------------------------------------------------------ verdict
\node[pass, text width=16.6cm, anchor=north] (v) at (8.6,-3.25) {%
  \scriptsize
  \textbf{accept} $\iff$ one identity across all three machines
  \ $\wedge$ \ every obligation re-earned rather than inherited
  \ $\wedge$ \ declared cost envelope holds\\[1pt]
  three-valued: \textsf{accept} / \textsf{reject} /
  \textsf{incomplete} --- a declared condition this run did not measure
  is \textsf{incomplete}, and \textsf{incomplete} never accepts};

% M3 drops straight into the verdict band; M2 routes down the right edge
\draw[flow] (12.9,-2.08) -- (12.9,-3.25);
\draw[flow] (m2.east) -- (16.65,1.55) -- (16.65,-3.25);

% ------------------------------------------------------------- caveat
\node[ev, anchor=north west, text width=16.6cm] at (0.3,-4.45) {%
  Neither leg establishes producer independence. The protocol records
  the vendor and model that produced M3 and marks same-vendor rebuilds
  \emph{independence unestablished}, but independence is a property of
  what a producer was exposed to, and no property of the emitted bytes
  decides it. A successful reconstruction is an observation, not a proof
  of prior non-exposure.};

\end{tikzpicture}
